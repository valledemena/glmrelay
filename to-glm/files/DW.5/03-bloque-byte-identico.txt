 567: // Fusion GIRO+DIAS para puntos (solo live; en replay solo GIRO con historia).
 568: // Regla: GIRO valido gana (primaria probada); DIAS rellena huecos con coords
 569: // resueltas contra el catalogo GIRO (el pager no trae lat/lon).
 570: static std::vector<StationSample> mergeDias(const std::vector<StationSample>& giro,
 571:                                             const std::vector<StationSample>& dias) {
 572:     std::vector<StationSample> out = giro;
 573:     for (const auto& d : dias) {
 574:         if (!d.valid) continue;
 575:         const StationSample* cat = nullptr;
 576:         bool covered = false;
 577:         for (const auto& g : giro) {
 578:             if (g.code != d.code) continue;
 579:             cat = &g;
 580:             if (g.valid) covered = true;
 581:             break;
 582:         }
 583:         if (covered || !cat) continue;
 584:         StationSample c = d;
 585:         c.latDeg = cat->latDeg;
 586:         c.lonDeg = cat->lonDeg;
 587:         c.name = cat->name;
 588:         if (c.latDeg == 0.0f && c.lonDeg == 0.0f) continue;  // sin posicion
 589:         out.push_back(c);
 590:     }
 591:     return out;
 592: }
